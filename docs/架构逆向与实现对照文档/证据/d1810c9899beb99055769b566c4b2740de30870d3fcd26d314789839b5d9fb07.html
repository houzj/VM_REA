	private bool CheckPassword(ComTriggerItem tiggerItem)
	{
		try
		{
			if (string.IsNullOrEmpty(tiggerItem.SolutionPath))
			{
				return true;
			}
			uint nHasPwd = 0u;
			uint nValidFormat = 0u;
			ServerSDKManager.serverSDKManager.mSolutionManager.CheckSolution(tiggerItem.SolutionPath, ref nHasPwd, ref nValidFormat);
			if (nValidFormat == 1 && nHasPwd == 1)
			{
				if (string.IsNullOrEmpty(tiggerItem.SolutionPasswd))
				{
					return true;
				}
				int nIsCorrect = -1;
				ServerSDKManager.serverSDKManager.mSolutionManager.CheckPassword(tiggerItem.SolutionPath, tiggerItem.SolutionPasswd, ref nIsCorrect);
				if (nIsCorrect == 0)
				{
					HikMessageBox.Show(AppLocalizationService.GetLang("方案密码错误"), MessageButtons.Normal, MessageIcons.Error);
					isSetChange = true;
					tiggerItem.SolutionPasswd = "";
					return true;
				}
			}
		}
		catch (Exception ex)
		{
			LogHelper.Error("CheckPassword is error." + ex.ToString());
			return false;
		}
		return true;
	}
