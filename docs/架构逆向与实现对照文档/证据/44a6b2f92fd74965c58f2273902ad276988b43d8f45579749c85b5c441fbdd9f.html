	private int SetTriggerItem(ComTriggerItem item, ActionType action)
	{
		try
		{
			if (DecorateSource == null)
			{
				return -1;
			}
			bool flag = item is GlobalTriggerEventItem;
			switch (action)
			{
			case ActionType.Add:
				(DecorateSource as VmModule).Params[flag ? "AddTriggerEvent" : "AddTriggerString"] = "true";
				break;
			case ActionType.Delete:
				(DecorateSource as VmModule).Params[flag ? "DeleteTriggerEvent" : "DeleteTriggerString"] = (item.Index - 1).ToString();
				break;
			case ActionType.Modify:
			{
				byte[] array = SetTriggerBytes(item, flag);
				if (array != null)
				{
					string key = (flag ? "ModifyTriggerEvent-" : "ModifyTriggerString-") + (item.Index - 1);
					try
					{
						((IUserBytesData)(DecorateSource as VmModule).Params)[key] = array;
					}
					catch (Exception ex)
					{
						LogHelper.Error("下发数据错误," + ex.ToString());
					}
				}
				break;
			}
			}
			return 0;
		}
		catch (Exception ex2)
		{
			if (ex2 is VmException ex3)
			{
				LogHelper.Error("下发数据错误," + ex3.errorMessage);
			}
			else
			{
				LogHelper.Error("下发数据错误," + ex2.ToString());
			}
			return -1;
		}
	}
